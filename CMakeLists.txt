cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "Static MSVC runtime")
project(SFTPAddIn VERSION 0.2.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "SFTPAddIn currently supports Windows with MSVC only")
endif()
if(NOT VCPKG_TARGET_TRIPLET MATCHES "^(x64|x86)-windows-static$")
    message(FATAL_ERROR "Use x64-windows-static or x86-windows-static")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_definitions(_WINDOWS UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/W4 /utf-8 /EHsc)
find_package(libssh2 CONFIG REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)

add_library(sftp_core STATIC src/SftpClient.cpp src/Utils.cpp)
target_include_directories(sftp_core PUBLIC src)
target_link_libraries(sftp_core PUBLIC libssh2::libssh2 OpenSSL::Crypto ws2_32)

add_library(SFTPAddIn SHARED src/AddIn.cpp src/Exports.cpp src/SFTPAddIn.def)
target_include_directories(SFTPAddIn SYSTEM PRIVATE third_party/1c)
target_link_libraries(SFTPAddIn PRIVATE sftp_core)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(SFTP_ARCH x64)
else()
    set(SFTP_ARCH x86)
endif()
set_target_properties(SFTPAddIn PROPERTIES OUTPUT_NAME "SFTPAddIn_${SFTP_ARCH}")
install(TARGETS SFTPAddIn RUNTIME DESTINATION .)
configure_file(packaging/info.xml.in "${CMAKE_CURRENT_BINARY_DIR}/info.xml" @ONLY)
configure_file(packaging/VerInfo.txt.in "${CMAKE_CURRENT_BINARY_DIR}/VerInfo.txt" @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/info.xml" "${CMAKE_CURRENT_BINARY_DIR}/VerInfo.txt" DESTINATION .)
install(DIRECTORY "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/libssh2/"
    DESTINATION licenses/libssh2 FILES_MATCHING PATTERN copyright)
install(DIRECTORY "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/openssl/"
    DESTINATION licenses/openssl FILES_MATCHING PATTERN copyright)
install(FILES third_party/1c/NOTICE.md third_party/1c/SecureStorage-LICENSE DESTINATION licenses/1c)

include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE sftp_core)
    add_test(NAME core_tests COMMAND core_tests)
    add_executable(native_host tests/native_host.cpp)
    target_include_directories(native_host SYSTEM PRIVATE third_party/1c)
    add_test(NAME native_api COMMAND native_host $<TARGET_FILE:SFTPAddIn>)
    add_executable(resource_stress tests/resource_stress.cpp)
    target_link_libraries(resource_stress PRIVATE sftp_core)
endif()
